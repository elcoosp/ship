+pub fn default_branch(shell: &dyn Shell, remote: &str) -> Result<String> {
+    let refname = format!("refs/remotes/{remote}/HEAD");
+    if let Ok(out) = shell.run(&["git", "symbolic-ref", "--short", &refname])
+        && out.status == 0
+    {
+        let raw = out.stdout.trim();
+        let stripped = raw.strip_prefix(&format!("{remote}/")).unwrap_or(raw);
+        if !stripped.is_empty() {
+            return Ok(stripped.to_string());
+        }
+    }
+    if let Ok(out) = shell.run(&["git", "ls-remote", "--symref", remote, "HEAD"])
+        && out.status == 0
+    {
+        for line in out.stdout.lines() {
+            if let Some(rest) = line.strip_prefix("ref: ")
+                && let Some((target, _)) = rest.split_once('\t')
+                && let Some(branch) = target.strip_prefix("refs/heads/")
+            {
+                return Ok(branch.to_string());
+            }
+        }
+    }
+    for candidate in ["main", "master"] {
+        let ok = shell
+            .run(&[
+                "git",
+                "rev-parse",
+                "--verify",
+                &format!("refs/heads/{candidate}"),
+            ])
+            .map(|o| o.status == 0)
+            .unwrap_or(false);
+        if ok {
+            return Ok(candidate.to_string());
+        }
+    }
+    bail!("could not resolve default branch for '{remote}' — try: git remote set-head {remote} -a")
+}

+pub fn is_git_repo_root(shell: &dyn Shell, expected: &Path) -> Result<bool> {
+    let out = match shell.run(&["git", "rev-parse", "--show-toplevel"]) {
+        Ok(o) if o.status == 0 => o,
+        _ => return Ok(false),
+    };
+    let top = PathBuf::from(out.stdout.trim());
+    let a = top.canonicalize().unwrap_or(top);
+    let b = expected
+        .canonicalize()
+        .unwrap_or_else(|_| expected.to_path_buf());
+    Ok(a == b)
+}

+pub fn union_check(dirty: &[PathBuf], manifest: &[PathBuf]) -> Result<()> {
+    let m: HashSet<&PathBuf> = manifest.iter().collect();
+    let missing: Vec<_> = dirty.iter().filter(|d| !m.contains(d)).collect();
+    if !missing.is_empty() {
+        let shown: Vec<_> = missing
+            .iter()
+            .take(10)
+            .map(|p| format!("    {}", p.display()))
+            .collect();
+        let extra = if missing.len() > 10 {
+            format!("\n    … and {} more", missing.len() - 10)
+        } else {
+            String::new()
+        };
+        bail!(
+            "ship.toml ↔ working tree mismatch\n  dirty but not in ship.toml ({}):\n{}{}\nAdd these paths to ship.toml.",
+            missing.len(),
+            shown.join("\n"),
+            extra
+        );
+    }
+    Ok(())

+pub fn no_git_operation_in_progress(shell: &dyn Shell) -> Result<()> {
+    let check = |name: &str| -> Result<bool> {
+        let out = run_checked(shell, &["git", "rev-parse", "--git-path", name])?;

+        "BISECT_LOG",

+        }
+    }
+    Ok(())

+pub fn find_git_locks(shell: &dyn Shell) -> Result<Vec<PathBuf>> {
+    let out = shell.run(&["git", "rev-parse", "--absolute-git-dir"])?;
+    if out.status != 0 {
+        return Ok(Vec::new());
+    }
+    let git_dir = PathBuf::from(out.stdout.trim());
+    let mut locks = Vec::new();
+
+    if let Ok(rd) = std::fs::read_dir(&git_dir) {
+        for e in rd.flatten() {
+            let p = e.path();
+            if p.extension().and_then(|s| s.to_str()) == Some("lock") {
+                locks.push(p);
+            }
+        }
+    }
+
+    fn walk(dir: &Path, out: &mut Vec<PathBuf>) {
+        if let Ok(rd) = std::fs::read_dir(dir) {
+            for e in rd.flatten() {
+                let p = e.path();
+                if p.is_dir() {
+                    walk(&p, out);
+                } else if p.extension().and_then(|s| s.to_str()) == Some("lock") {
+                    out.push(p);
+                }
+            }
+        }
+    }
+    walk(&git_dir.join("refs"), &mut locks);
+    Ok(locks)
+}

+pub fn ensure_no_git_locks(shell: &dyn Shell, clean_stale: bool) -> Result<()> {
+    for attempt in 0..20 {
+        let locks = find_git_locks(shell)?;
+        if locks.is_empty() {
+            return Ok(());
+        }
+
+        let now = std::time::SystemTime::now();
+        let mut stale: Vec<(PathBuf, u64)> = Vec::new();
+        let mut fresh: Vec<PathBuf> = Vec::new();
+        for p in locks {
+            let age = std::fs::metadata(&p)
+                .ok()
+                .and_then(|m| m.modified().ok())
+                .and_then(|t| now.duration_since(t).ok());
+            match age {
+                Some(d) if d.as_secs() >= 30 => stale.push((p, d.as_secs())),
+                _ => fresh.push(p),
+            }
+        }
+
+        if !stale.is_empty() {
+            if clean_stale {
+                for (p, age) in &stale {
+                    let _ = std::fs::remove_file(p);
+                    eprintln!("removed stale lock (age {age}s): {}", p.display());
+                }
+            } else {
+                return Err(format_locks_err(&stale, &fresh));
+            }
+        }
+
+        if attempt == 19 {
+            return Err(format_locks_err(&[], &fresh));
+        }
+        std::thread::sleep(std::time::Duration::from_millis(500));
+    }
+    Ok(())
+}
+
+fn format_locks_err(stale: &[(PathBuf, u64)], fresh: &[PathBuf]) -> anyhow::Error {
+    let mut msg = String::from("git lock files present:\n");
+    for (p, age) in stale {
+        msg.push_str(&format!("  STALE ({}s old): {}\n", age, p.display()));
+    }
+    for p in fresh {
+        msg.push_str(&format!(
+            "  FRESH (<30s, retried 10s without clearing): {}\n",
+            p.display()
+        ));
+    }
+    msg.push_str("\nRecovery:\n");
+    if !stale.is_empty() {
+        msg.push_str("  • Re-run with --clean-locks to remove stale locks automatically\n");
+    }
+    msg.push_str(
+        "  • Or manually remove each lock file above (safe if no git process is running)\n",
+    );
+    msg.push_str(
+        "  • If locks persist: an editor's git integration may be holding them; close it and retry\n",
+    );
+    anyhow::anyhow!("{}", msg)
+}

+pub fn ship_artifacts_gitignored(shell: &dyn Shell, manifest: &Path, pr_body: &Path) -> Result<()> {
+    let mut hard_fail: Vec<String> = Vec::new();
+    let mut soft_warn: Vec<String> = Vec::new();
+    let mut to_append: Vec<String> = Vec::new();
+
+    for path in [manifest, pr_body] {
+        let s = path.to_string_lossy().into_owned();
+        let tracked = run_ok(shell, &["git", "ls-files", "--error-unmatch", &s])?;
+        let ignored = run_ok(shell, &["git", "check-ignore", "-q", &s])?;
+
+        if ignored {
+            if tracked {
+                soft_warn.push(format!(
+                    concat!(
+                        "{} is both tracked AND matched by .gitignore.\n",
+                        "This usually means .gitignore was added after the file was already committed.\n",
+                        "Consider: git rm --cached {}"
+                    ),
+                    path.display(),
+                    s
+                ));
+            }
+            continue;
+        }
+
+        if tracked {
+            hard_fail.push(format!(
+                concat!(
+                    "{} is TRACKED (in git's index).\n",
+                    ".gitignore only applies to untracked files, so adding a pattern does NOT help here.\n",
+                    "\n",
+                    "Fix — pick one:\n",
+                    "  A. Untrack it (recommended): git rm --cached {s}\n",
+                    "     Then .gitignore applies. The file stays on disk and stays in .gitignore.\n",
+                    "  B. Keep tracking it — remove its entry from .gitignore, accept it in the tree,\n",
+                    "     and either add it to ship.toml or remove it from the repo before shipping.\n",
+                    "\n",
+                    "Verify current state:\n",
+                    "  git ls-files {s}      # shows it if tracked\n",
+                    "  git check-ignore {s}  # silent + exit 0 if ignored, exit 1 if not"
+                ),
+                path.display(),
+                s = s
+            ));
+        } else {
+            to_append.push(s);
+        }
+    }
+
+    if !to_append.is_empty() {
+        let gi = Path::new(".gitignore");
+        let mut content = if gi.exists() {
+            std::fs::read_to_string(gi).unwrap_or_default()
+        } else {
+            String::new()
+        };
+        if !content.is_empty() && !content.ends_with('\n') {
+            content.push('\n');
+        }
+        for line in &to_append {
+            content.push_str(line);
+            content.push('\n');
+            eprintln!("WARN: added '{line}' to .gitignore (was missing)");
+        }
+        std::fs::write(gi, content)?;
+
+        for line in &to_append {
+            let ok = run_ok(shell, &["git", "check-ignore", "-q", line])?;
+            if !ok {
+                hard_fail.push(format!(
+                    "{} is still not gitignored after appending — check .gitignore manually",
+                    line
+                ));
+            }
+        }
+    }
+
+    for w in &soft_warn {
+        eprintln!("WARN: {w}");
+    }
+    if !hard_fail.is_empty() {
+        bail!("{}", hard_fail.join("\n\n"));
+    }
+    Ok(())
+}

+use ship::manifest;
+use ship::pipeline::{self, Outcome};

+    cmd: Cmd,

+#[derive(Subcommand)]
+enum Cmd {
+    Prepare {
+        #[arg(long)]
+        dry_run: bool,
+        #[arg(long)]
+        reset: bool,
+        /// Remove stale (>30s old) git lock files before running.
+        #[arg(long)]
+        clean_locks: bool,
+    },
+    Tag {
+        #[arg(long)]
+        dry_run: bool,
+        #[arg(long)]
+        clean_locks: bool,
+    },
+    All {
+        #[arg(long)]
+        dry_run: bool,
+        #[arg(long)]
+        reset: bool,
+        #[arg(long)]
+        clean_locks: bool,
+    },
+    Watch {
+        #[arg(long)]
+        watch_dir: Option<PathBuf>,
+        #[arg(long)]
+        repos_root: Option<PathBuf>,
+        #[arg(long)]
+        log_file: Option<PathBuf>,
+        #[arg(long)]
+        once: bool,
+        #[arg(long)]
+        allow_repo_manifest: bool,
+        #[arg(long)]
+        clean_locks: bool,
+    },
+    Projects {
+        #[arg(long)]
+        repos_root: Option<PathBuf>,
+    },
+}

+fn main() -> anyhow::Result<()> {
+    let cli = Cli::parse();
+
+    // Init the log sink for every subcommand. Watch overrides with its own
+    // --log-file when present (OnceLock::set is idempotent — first wins, so
+    // watch's init happens inside watch::run BEFORE any other path).
+    let default_log = std::env::var("HOME")
+        .ok()
+        .map(|h| PathBuf::from(h).join(".local/state/ship/watcher.log"));
+    ship::ui::init_log(default_log);
+
+    if let Cmd::Watch {
+        watch_dir,
+        repos_root,
+        log_file,
+        once,
+        allow_repo_manifest,
+        clean_locks,
+    } = &cli.cmd
+    {
+        let mut cfg = WatchConfig::default();
+        if let Some(p) = watch_dir {
+            cfg.watch_dir = p.clone();
+        }
+        if let Some(p) = repos_root {
+            cfg.repos_root = p.clone();
+        }
+        if let Some(p) = log_file {
+            cfg.log_file = Some(p.clone());
+        }
+        cfg.allow_repo_manifest = *allow_repo_manifest;
+        cfg.clean_locks = *clean_locks;
+        return watch::run(cfg, *once);
+    }
+
+    if let Cmd::Projects { repos_root } = &cli.cmd {
+        let root = repos_root.clone().unwrap_or_else(watch::default_repos_root);
+        let projects = watch::list_projects(&root)?;
+        ship::ui::projects_table(&root, &projects);
+        return Ok(());
+    }
+
+    let manifest = manifest::Manifest::load(&cli.manifest)?;
+    let shell = RealShell::new(std::env::current_dir()?);
+
+    let outcome = match cli.cmd {
+        Cmd::Prepare {
+            dry_run,
+            reset,
+            clean_locks,
+        } => pipeline::prepare(dry_run, reset, clean_locks, &manifest, &shell)?,
+        Cmd::Tag {
+            dry_run,
+            clean_locks,
+        } => pipeline::tag(dry_run, clean_locks, &manifest, &shell)?,
+        Cmd::All {
+            dry_run,
+            reset,
+            clean_locks,
+        } => {
+            let p = pipeline::prepare(dry_run, reset, clean_locks, &manifest, &shell)?;
+            if dry_run {
+                p
+            } else {
+                let t = pipeline::tag(false, clean_locks, &manifest, &shell)?;
+                Outcome {
+                    pr_url: p.pr_url.or(t.pr_url),
+                    merged: p.merged.or(t.merged),
+                    tag: t.tag,
+                }
+            }
+        }
+        Cmd::Watch { .. } | Cmd::Projects { .. } => unreachable!(),
+    };
+
+    if cli.cmd_is_prepare_like() {
+        ship::ui::final_success(
+            outcome.pr_url.as_deref(),
+            outcome.merged.as_deref(),
+            outcome.tag.as_deref(),
+        );
+    } else {
+        // tag command: no PR panel, just the tag
+        if let Some(t) = outcome.tag.as_deref() {
+            ship::ui::tagged(t, true);
+        }
+    }
+    Ok(())
+}

+impl Cli {
+    fn cmd_is_prepare_like(&self) -> bool {
+        matches!(self.cmd, Cmd::Prepare { .. } | Cmd::All { .. })
+    }

+        let m: Manifest = toml::from_str(&text).context("parsing ship.toml")?;
+        m.validate()?;
+        Ok(m)
+    }
+
+    fn validate(&self) -> Result<()> {
+        let prefixes = [
+            "feat", "fix", "docs", "refactor", "perf", "chore", "build", "test", "ci", "style",
+            "revert",
+        ];
+        let (prefix, _) = self
+            .delivery
+            .branch
+            .split_once('/')
+            .with_context(|| format!("branch '{}' has no prefix/", self.delivery.branch))?;
+        if !prefixes.contains(&prefix) {
+            bail!("branch prefix '{}' not in {:?}", prefix, prefixes);
+        }
+        if self.commit.is_empty() {
+            bail!("manifest has no [[commit]] groups");
+        }
+        let re = Regex::new(
+            r"^(feat|fix|docs|refactor|perf|chore|build|test|ci|style|revert)(\([^)]+\))?!?: .+$",
+        )?;
+        let mut seen: HashSet<&PathBuf> = HashSet::new();
+        for (i, g) in self.commit.iter().enumerate() {
+            let subject = g.message.lines().next().unwrap_or("");
+            if subject.len() > 72 {
+                bail!(
+                    "[[commit]][{}] subject is {} chars (>72): {:?}",
+                    i,
+                    subject.len(),
+                    subject
+                );
+            }
+            if !re.is_match(subject) {
+                bail!(
+                    "[[commit]][{}] not a conventional subject: {:?}",
+                    i,
+                    subject
+                );
+            }
+            for f in &g.files {
+                if !seen.insert(f) {
+                    bail!(
+                        "path {} appears in more than one [[commit]] group",
+                        f.display()
+                    );
+                }
+            }
+        }
+        Ok(())
+    }
+
+    /// Expand every `files` entry against the actual dirty set.
+    ///
+    /// Rules:
+    ///   • Literal path that IS dirty → included.
+    ///   • Literal path that is NOT dirty → dropped (recorded in DropReport).
+    ///     Resume-friendly: files already committed or missing from the zip
+    ///     are skipped, not fatal.
+    ///   • Glob that matches ≥1 dirty path → included.
+    ///   • Glob that matches 0 dirty paths → dropped.
+    ///   • Invalid glob syntax → fatal (typo in the pattern).
+    ///   • Overlap across groups after expansion → fatal.
+    ///
+    /// A group with zero remaining files is skipped by the caller.
+    pub fn expand_globs(&self, dirty: &[PathBuf]) -> Result<(Self, DropReport)> {
+        let dirty_set: HashSet<&PathBuf> = dirty.iter().collect();
+        let mut out = self.clone();
+        let mut report = DropReport {
+            per_group: vec![Vec::new(); self.commit.len()],
+        };
+
+        for (i, group) in out.commit.iter_mut().enumerate() {
+            let mut expanded: Vec<PathBuf> = Vec::new();
+            for pattern in &group.files {
+                let s = pattern.to_string_lossy();
+                if !has_glob_chars(&s) {
+                    if dirty_set.contains(pattern) {
+                        expanded.push(pattern.clone());
+                    } else {
+                        report.per_group[i].push(pattern.clone());
+                    }
+                    continue;
+                }
+                let matcher = GlobBuilder::new(&s)
+                    .literal_separator(true)
+                    .build()
+                    .with_context(|| {
+                        format!("invalid glob in [[commit]] '{}': {}", group.message, s)
+                    })?
+                    .compile_matcher();
+                let before = expanded.len();
+                for d in dirty {
+                    if matcher.is_match(d) {
+                        expanded.push(d.clone());
+                    }
+                }
+                if expanded.len() == before {
+                    report.per_group[i].push(pattern.clone());
+                }
+            }
+            expanded.sort();
+            expanded.dedup();
+            group.files = expanded;
+        }
+
+        // Overlap after expansion (two globs hitting the same dirty file).
+        let mut seen: HashSet<PathBuf> = HashSet::new();
+        for (i, g) in out.commit.iter().enumerate() {
+            for f in &g.files {
+                if !seen.insert(f.clone()) {
+                    bail!(
+                        "path {} appears in multiple [[commit]] groups after glob expansion (check [[commit]][{i}] = {:?})",
+                        f.display(),
+                        g.message
+                    );
+                }
+            }
+        }
+        Ok((out, report))
+    }
+}
+
+fn has_glob_chars(s: &str) -> bool {
+    s.contains('*') || s.contains('?') || s.contains('[')
+}

+use crate::shell::Shell;
+use anyhow::Result;
+use std::os::unix::fs::PermissionsExt;
+
+/// Revert spurious mode changes on tracked files.
+///
+/// Reads the exact mode from the Git index (`git ls-files -s`), checks the
+/// actual file on disk, and forcibly applies `chmod` to match the index.
+/// This bypasses `git diff` entirely, making it immune to format differences.
+pub fn exec_bit(shell: &dyn Shell) -> Result<usize> {
+    let out = shell.run(&["git", "ls-files", "-s"])?;
+    if out.status != 0 {
+        return Ok(0);
+    }
+
+    let cwd = shell.cwd();
+    let mut reverted = 0usize;
+
+    for line in out.stdout.lines() {
+        // Format: 100644 <sha> 0\tpath/to/file
+        let parts: Vec<&str> = line.split('\t').collect();
+        if parts.len() != 2 {
+            continue;
+        }
+        let meta = parts[0];
+        let path = parts[1];
+
+        let meta_parts: Vec<&str> = meta.split(' ').collect();
+        if meta_parts.len() < 3 {
+            continue;
+        }
+
+        // Parse the mode from the index (e.g., 100755 -> 33261)
+        let git_mode = u32::from_str_radix(meta_parts[0], 8).unwrap_or(0);
+
+        let full_path = cwd.join(path);
+        if let Ok(metadata) = std::fs::metadata(&full_path) {
+            let disk_mode = metadata.permissions().mode();
+
+            // Check if the executable bit matches
+            let want_exec = git_mode & 0o111 != 0;
+            let has_exec = disk_mode & 0o111 != 0;
+
+            if want_exec != has_exec {
+                let mut perms = metadata.permissions();
+                if want_exec {
+                    perms.set_mode(0o755);
+                } else {
+                    perms.set_mode(0o644);
+                }
+                let _ = std::fs::set_permissions(&full_path, perms);
+                reverted += 1;
+            }
+        }
+    }
+    Ok(reverted)
+}

+pub fn tag(dry_run: bool, clean_locks: bool, m: &Manifest, shell: &dyn Shell) -> Result<Outcome> {
+    let remote = &m.delivery.remote;
+    let branch = &m.delivery.branch;
+    let default = guards::default_branch(shell, remote)?;
+    guards::gh_authenticated(shell)?;
+    guards::ensure_no_git_locks(shell, clean_locks)?;

+        .ok_or_else(|| anyhow!("no merged PR for {branch} — run `ship prepare` first"))?;

+        let version = match version::read_workspace_version_at_opt(
+            shell,
+            &format!("{remote}/{default}"),
+            "Cargo.toml",

+            Some(v) => v,
+            None => version::read_workspace_version(&shell.cwd().join("Cargo.toml"))?,
+        };

+    let head = run_checked(shell, &["git", "rev-parse", "HEAD"])?

+fn lookup_merged(shell: &dyn Shell, branch: &str) -> Result<Option<MergeInfo>> {
+    let out = run_checked(
+        shell,
+        &[
+            "gh",
+            "pr",
+            "list",
+            "--head",
+            branch,
+            "--state",
+            "merged",
+            "--json",
+            "url,mergeCommit",
+            "--jq",
+            r#".[0] | select(. != null) | "\(.url) \(.mergeCommit.oid)""#,
+        ],
+    )?
+    .stdout;
+    let line = out.trim();
+    if line.is_empty() {
+        return Ok(None);
+    }
+    let mut it = line.splitn(2, ' ');
+    let url = it.next().context("url")?.to_string();
+    let sha = it.next().context("sha")?.to_string();
+    Ok(Some(MergeInfo { url, sha }))

+fn merge_immediately(shell: &dyn Shell, pr_url: &str) -> Result<String> {
+    let state = pr_state(shell, pr_url)?;
+    match state.as_str() {
+        "MERGED" => {
+            return merged_sha(shell, pr_url)?
+                .ok_or_else(|| anyhow!("PR is MERGED but mergeCommit.oid is empty"));
+        }
+        "CLOSED" => bail!("PR {pr_url} is CLOSED — not merging"),
+        _ => {}
+    }
+    crate::ui::step("merging PR (--admin)...");
+    let out = shell.run(&[
+        "gh",
+        "pr",
+        "merge",
+        pr_url,
+        "--admin",
+        "--rebase",
+        "--delete-branch",
+    ])?;
+    if out.status != 0 {
+        bail!(
+            "gh pr merge --admin failed (are you a repo admin?): {}",
+            out.stderr.trim()
+        );
+    }
+    for attempt in 0..3 {
+        if let Some(sha) = merged_sha(shell, pr_url)? {
+            return Ok(sha);
+        }
+        if attempt < 2 {
+            std::thread::sleep(Duration::from_secs(5));
+        }
+    }
+    bail!("gh pr merge --admin returned 0 but mergeCommit.oid is still empty after 10s")
+}

+fn ensure_on_branch(shell: &dyn Shell, m: &Manifest, remote: &str, default: &str) -> Result<()> {
+    let branch = &m.delivery.branch;
+    run_checked(shell, &["git", "fetch", remote, "--prune"])?;
+    let current = run_checked(shell, &["git", "rev-parse", "--abbrev-ref", "HEAD"])?
+        .stdout
+        .trim()
+        .to_string();
+    if &current == branch {
+        return Ok(());
+    }
+    let stashed = guards::is_dirty(shell)?;
+    if stashed {
+        run_checked(
+            shell,
+            &["git", "stash", "push", "-u", "-m", "ship: pre-branch"],
+        )?;
+    }
+    let switch_res = run_checked(
+        shell,
+        &[
+            "git",
+            "switch",
+            "-C",
+            branch,
+            &format!("{remote}/{default}"),
+        ],
+    );
+    if let Err(e) = switch_res {
+        if e.to_string().contains("index.lock") {
+            crate::ui::warn("git index is locked, retrying switch...");
+            std::thread::sleep(std::time::Duration::from_millis(1000));
+            run_checked(
+                shell,
+                &[
+                    "git",
+                    "switch",
+                    "-C",
+                    branch,
+                    &format!("{remote}/{default}"),
+                ],
+            )?;
+        } else {
+            return Err(e);
+        }
+    }
+    if stashed {
+        run_checked(shell, &["git", "stash", "pop"])?;
+    }
+    let now = run_checked(shell, &["git", "rev-parse", "--abbrev-ref", "HEAD"])?
+        .stdout
+        .trim()
+        .to_string();
+    if &now != branch {
+        bail!("expected to be on {branch}, got {now}");
+    }
+    Ok(())
+}

+fn ensure_committed(shell: &dyn Shell, m: &Manifest, remote: &str, default: &str) -> Result<()> {
+    let base = format!("{remote}/{default}");
+    let total = m.commit.len();
+    for (idx, g) in m.commit.iter().enumerate() {
+        if g.files.is_empty() {
+            continue;
+        }
+        if already_committed(shell, &g.message, &base)? {
+            continue;
+        }
+        let paths: Vec<String> = g
+            .files
+            .iter()
+            .map(|f| f.to_string_lossy().into_owned())
+            .collect();
+        let mut add: Vec<&str> = vec!["git", "add", "-A", "--"];
+        for p in &paths {
+            add.push(p);
+        }
+        run_checked(shell, &add)?;
+
+        // git status --porcelain is the absolute truth. If it's empty, skip.
+        let status_out = shell.run(&["git", "status", "--porcelain"])?;
+        if status_out.stdout.trim().is_empty() {
+            crate::ui::warn(&format!(
+                "git status is clean, skipping commit for '{}'",
+                g.message
+            ));
+            continue;
+        }
+
+        let commit_res = shell.run(&["git", "commit", "--no-verify", "-m", &g.message]);
+        if let Ok(out) = &commit_res
+            && out.status != 0
+        {
+            eprintln!(
+                "
+--- GIT COMMIT FAILED ---"
+            );
+            eprintln!("Command: git commit --no-verify -m {:?}", g.message);
+            eprintln!("Exit code: {}", out.status);
+            eprintln!(
+                "Stdout:
+{}",
+                out.stdout
+            );
+            eprintln!(
+                "Stderr:
+{}",
+                out.stderr
+            );
+            eprintln!(
+                "Git Status:
+{}",
+                status_out.stdout
+            );
+            eprintln!(
+                "-------------------------
+"
+            );
+        }
+        commit_res?;
+        crate::ui::commit_step(idx + 1, total, &g.message);
+    }
+    Ok(())
+}

+    guards::local_default_not_ahead(shell, remote, default)?;
+    let stashed = guards::is_dirty(shell)?;
+    if stashed {
+        run_checked(
+            shell,
+            &["git", "stash", "push", "-u", "-m", "ship: pre-sync"],
+        )?;
+    }
+    run_checked(
+        shell,
+        &[
+            "git",
+            "switch",
+            "-C",
+            default,
+            &format!("{remote}/{default}"),
+        ],
+    )?;
+    if stashed {
+        run_checked(shell, &["git", "stash", "pop"])?;
+    }
+    let head = run_checked(shell, &["git", "rev-parse", "HEAD"])?

+        .trim()
+        .to_string();
+    if head != merged {
+        bail!("after sync, HEAD is {head} but the merged SHA was {merged} — origin moved mid-run");
+    }
+    Ok(())

+use anyhow::{Context, Result, bail};
+use std::path::PathBuf;
+use std::process::Command;
+
+#[derive(Clone)]
+pub struct Output {
+    pub status: i32,
+    pub stdout: String,
+    pub stderr: String,
+}
+
+pub trait Shell {
+    fn run(&self, argv: &[&str]) -> Result<Output>;
+    fn cwd(&self) -> &std::path::Path;
+}
+
+/// Runs commands in `cwd`. Never mutates process state beyond spawning.
+pub struct RealShell {
+    cwd: PathBuf,
+}
+
+impl RealShell {
+    pub fn new(cwd: PathBuf) -> Self {
+        Self { cwd }
+    }
+}
+
+impl Shell for RealShell {
+    fn cwd(&self) -> &std::path::Path {
+        &self.cwd
+    }
+    fn run(&self, argv: &[&str]) -> Result<Output> {
+        if argv.is_empty() {
+            bail!("empty argv");
+        }
+        let out = Command::new(argv[0])
+            .args(&argv[1..])
+            .current_dir(&self.cwd)
+            .output()
+            .with_context(|| format!("executing {:?}", argv))?;
+        Ok(Output {
+            status: out.status.code().unwrap_or(-1),
+            stdout: String::from_utf8_lossy(&out.stdout).into_owned(),
+            stderr: String::from_utf8_lossy(&out.stderr).into_owned(),
+        })
+    }
+}
+
+/// Fail on non-zero exit. `argv` is shown in the error message.
+pub fn run_checked(shell: &dyn Shell, argv: &[&str]) -> Result<Output> {
+    let out = shell.run(argv)?;
+    if out.status != 0 {
+        bail!("command failed: {}\n{}", argv.join(" "), out.stderr.trim());
+    }
+    Ok(out)
+}
+
+/// Success/failure without an error on failure (for `git rev-parse --verify`).
+pub fn run_ok(shell: &dyn Shell, argv: &[&str]) -> Result<bool> {
+    Ok(shell.run(argv)?.status == 0)
+}

+pub fn commit_step(idx: usize, total: usize, subject: &str) {
+    let counter = format!("[{:>2}/{}]", idx, total);
+
+    let (p_str, rest) = if let Some(pos) = subject.find(':') {
+        let (p, r) = subject.split_at(pos);
+        (p.trim(), r[1..].trim())
+    } else {
+        ("", subject)
+    };
+
+    let colored_prefix = if p_str.starts_with("feat") {
+        Span::new(p_str).fg(C_SUCCESS)
+    } else if p_str.starts_with("fix") {
+        Span::new(p_str).fg(C_ERROR)
+    } else if p_str.starts_with("docs") {
+        Span::new(p_str).fg(C_PRIMARY)
+    } else {
+        Span::new(p_str).dim()
+    };
+
+    let mut line = Line::new()
+        .push(padded_prefix())
+        .fg(C_DIM)
+        .push("✓ ")
+        .fg(C_SUCCESS)
+        .dim()
+        .push(counter.as_str())
+        .fg(C_DIM)
+        .push("  ");
+
+    if !p_str.is_empty() {
+        if let Some(scope_end) = rest.find(')') {
+            let (scope, msg) = rest.split_at(scope_end + 1);
+            line = line
+                .push(colored_prefix)
+                .push(scope)
+                .dim()
+                .push(msg)
+                .white();
+        } else {
+            line = line
+                .push(colored_prefix)
+                .push(": ")
+                .white()
+                .push(rest)
+                .white();
+        }
+    } else {
+        line = line.push(rest).white();
+    }
+    print_line(&line);
+    step(&format!("committed {}", counter));
+}

+pub enum BumpLevel {

+    Minor,
+    Major,

+pub fn derive_bump(m: &Manifest) -> BumpLevel {
+    let mut level = BumpLevel::Patch;
+    for g in &m.commit {
+        let subject = g.message.lines().next().unwrap_or("");
+        if subject.contains("!:") || g.message.contains("BREAKING CHANGE:") {
+            return BumpLevel::Major;
+        }
+        if subject.starts_with("feat") {
+            level = BumpLevel::Minor;
+        }
+    }
+    level
+}

+#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd)]
+pub struct Version {
+    pub major: u64,
+    pub minor: u64,
+    pub patch: u64,
+}
+
+impl Version {
+    pub fn parse(s: &str) -> Result<Self> {
+        let core = s.split('-').next().unwrap_or(s);
+        let mut it = core.split('.');
+        let major = it.next().context("version major")?.parse()?;
+        let minor = it.next().context("version minor")?.parse()?;
+        let patch = it.next().context("version patch")?.parse()?;
+        Ok(Self {
+            major,
+            minor,
+            patch,
+        })
+    }
+
+    pub fn bump(&self, level: BumpLevel) -> Self {
+        match level {
+            BumpLevel::Major => Self {
+                major: self.major + 1,
+                minor: 0,
+                patch: 0,
+            },
+            BumpLevel::Minor => Self {
+                major: self.major,
+                minor: self.minor + 1,
+                patch: 0,
+            },
+            BumpLevel::Patch => Self {
+                major: self.major,
+                minor: self.minor,
+                patch: self.patch + 1,
+            },
+        }
+    }
+}
+
+impl std::fmt::Display for Version {
+    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
+        write!(f, "{}.{}.{}", self.major, self.minor, self.patch)
+    }

+    let home = std::env::var("HOME").unwrap_or_else(|_| ".".into());
+    let candidates = [
+        PathBuf::from(&home).join("Documents/Repos"),
+        PathBuf::from(&home).join("Documents/repos"),
+        PathBuf::from(&home).join("Repos"),
+        PathBuf::from(&home).join("repos"),
+    ];
+    for c in &candidates {
+        if c.is_dir() {
+            return c.clone();
+        }
+    }
+    candidates[0].clone()
+}

+pub fn list_projects(repos_root: &Path) -> Result<Vec<String>> {
+    let read = std::fs::read_dir(repos_root)
+        .with_context(|| format!("cannot read repos_root {}", repos_root.display()))?;
+    let mut projects = Vec::new();
+    for entry in read.flatten() {
+        if !entry.file_type().map(|t| t.is_dir()).unwrap_or(false) {
+            continue;
+        }
+        if let Some(name) = entry.file_name().to_str() {
+            projects.push(name.to_string());
+        }
+    }
+    projects.sort();
+    Ok(projects)
+}
+
+fn clean_old_processed(dir: &Path) {
+    if let Ok(entries) = std::fs::read_dir(dir) {
+        let now = std::time::SystemTime::now();
+        for entry in entries.flatten() {
+            if let Ok(meta) = entry.metadata()
+                && let Ok(modified) = meta.modified()
+                && let Ok(age) = now.duration_since(modified)
+                && age.as_secs() > 7 * 24 * 3600
+            {
+                let _ = std::fs::remove_file(entry.path());
+            }
+        }
+    }
+}

+pub fn run(cfg: WatchConfig, once: bool) -> Result<()> {
+    std::fs::create_dir_all(&cfg.watch_dir).context("create watch_dir")?;
+    std::fs::create_dir_all(&cfg.processed_dir)?;
+    clean_old_processed(&cfg.processed_dir);
+    std::fs::create_dir_all(&cfg.failed_dir)?;
+    if let Some(p) = &cfg.log_file
+        && let Some(parent) = p.parent()
+    {
+        std::fs::create_dir_all(parent)?;
+    }
+    ui::init_log(cfg.log_file.clone());
+    let _lock = acquire_lock(&cfg.watch_dir.join(".ship-watcher.pid"))?;
+
+    ui::banner_startup();
+
+    let projects = list_projects(&cfg.repos_root).unwrap_or_default();
+    let mut flags: Vec<&str> = Vec::new();
+    if cfg.clean_locks {
+        flags.push("clean_locks");
+    }
+    if cfg.allow_repo_manifest {
+        flags.push("allow_repo_manifest");
+    }
+    if cfg.poll_only {
+        flags.push("poll");
+    }
+
+    ui::context(&cfg.watch_dir, &cfg.repos_root, &projects, &flags);
+    if projects.is_empty() {
+        ui::warn("no projects found — pass --repos-root DIR or set SHIP_REPOS_ROOT");
+    }
+    ui::ready();
+
+    let start = Instant::now();
+    let shipped = Arc::new(AtomicUsize::new(0));
+    let failed = Arc::new(AtomicUsize::new(0));
+    let mut seen: HashSet<PathBuf> = HashSet::new();
+
+    let zips = list_zips(&cfg.watch_dir)?;
+    if !zips.is_empty() {
+        for zip in zips {
+            seen.insert(zip.clone());
+            process_zip(&cfg, &zip, Arc::clone(&shipped), Arc::clone(&failed));
+        }
+        if once {
+            let mut waited = 0;
+            while ACTIVE_COUNT.load(Ordering::Relaxed) > 0 && waited < 300 {
+                std::thread::sleep(Duration::from_millis(500));
+                waited += 1;
+            }
+            let s = shipped.load(Ordering::Relaxed);
+            let f = failed.load(Ordering::Relaxed);
+            if s > 0 || f > 0 {
+                ui::final_summary(s, f, start.elapsed());
+            }
+        }
+        if once {
+            return Ok(());
+        }
+    }
+
+    if cfg.poll_only {
+        ui::info(&format!(
+            "poll mode active (interval {:?}) — drop a zip to ship",
+            cfg.poll_interval
+        ));
+        return poll_loop(&cfg, seen, shipped, failed);
+    }
+
+    let (tx, rx) = channel();
+    match notify::recommended_watcher(move |ev| {
+        let _ = tx.send(ev);
+    }) {
+        Ok(mut watcher) => match watcher.watch(&cfg.watch_dir, RecursiveMode::NonRecursive) {
+            Ok(()) => return debounce_loop(&cfg, rx, seen, shipped, failed),
+            Err(e) => ui::warn(&format!(
+                "notify watch failed ({e}) — falling back to polling"
+            )),
+        },
+        Err(e) => ui::warn(&format!(
+            "notify init failed ({e}) — falling back to polling"
+        )),
+    }
+    poll_loop(&cfg, seen, shipped, failed)
+}
+
+fn poll_loop(
+    cfg: &WatchConfig,
+    mut seen: HashSet<PathBuf>,
+    shipped: Arc<AtomicUsize>,
+    failed: Arc<AtomicUsize>,
+) -> Result<()> {
+    loop {
+        match list_zips(&cfg.watch_dir) {
+            Ok(zips) => {
+                for zip in zips {
+                    if seen.contains(&zip) {
+                        continue;
+                    }
+                    if !zip.exists() {
+                        seen.insert(zip);
+                        continue;
+                    }
+                    seen.insert(zip.clone());
+                    process_zip(cfg, &zip, Arc::clone(&shipped), Arc::clone(&failed));
+                }
+            }
+            Err(e) => ui::warn(&format!("list_zips failed: {e:#}")),
+        }
+        std::thread::sleep(cfg.poll_interval);
+    }
+}
+
+fn debounce_loop(
+    cfg: &WatchConfig,
+    rx: Receiver<notify::Result<notify::Event>>,
+    mut seen: HashSet<PathBuf>,
+    shipped: Arc<AtomicUsize>,
+    failed: Arc<AtomicUsize>,
+) -> Result<()> {
+    let mut pending: HashMap<PathBuf, Instant> = HashMap::new();
+    loop {
+        if let Ok(Ok(event)) = rx.recv_timeout(Duration::from_millis(200)) {
+            let relevant = matches!(event.kind, EventKind::Create(_) | EventKind::Modify(_));
+            if !relevant {
+                continue;
+            }
+            for p in event.paths {
+                if !is_zip(&p) {
+                    continue;
+                }
+                if p.parent() != Some(cfg.watch_dir.as_path()) {
+                    continue;
+                }
+                if p.file_name()
+                    .map(|n| n.to_string_lossy().starts_with('.'))
+                    .unwrap_or(false)
+                {
+                    continue;
+                }
+                pending.insert(p, Instant::now());
+            }
+        }
+        let now = Instant::now();
+        let settled: Vec<PathBuf> = pending
+            .iter()
+            .filter(|(_, t)| now.duration_since(**t) >= cfg.settle)
+            .map(|(p, _)| p.clone())
+            .collect();
+        for zip in settled {
+            pending.remove(&zip);
+            if seen.contains(&zip) {
+                continue;
+            }
+            if !zip.exists() {
+                seen.insert(zip);
+                continue;
+            }
+            seen.insert(zip.clone());
+            process_zip(cfg, &zip, Arc::clone(&shipped), Arc::clone(&failed));
+        }
+    }
+}
+
+static ACTIVE_REPOS: Mutex<Option<HashSet<PathBuf>>> = Mutex::new(None);
+static ACTIVE_COUNT: AtomicUsize = AtomicUsize::new(0);
+
+fn process_zip(
+    cfg: &WatchConfig,
+    zip: &Path,
+    shipped: Arc<AtomicUsize>,
+    failed: Arc<AtomicUsize>,
+) -> bool {
+    if !zip.exists() {
+        return false;
+    }
+    let name = zip
+        .file_name()
+        .map(|n| n.to_string_lossy().into_owned())
+        .unwrap_or_else(|| zip.display().to_string());
+    let stem = match zip.file_stem().and_then(|s| s.to_str()) {
+        Some(s) => s.to_string(),
+        None => {
+            ui::skip(&format!("{name}: bad filename"));
+            return false;
+        }
+    };
+
+    let project = match resolve_project(&cfg.repos_root, &stem) {
+        Ok(Some((p, _slug))) => p,
+        Ok(None) => {
+            ui::skip(&format!("{name}: no project matches '{stem}'"));
+            return false;
+        }
+        Err(e) => {
+            ui::error_panel(&name, &format!("{e:#}"), &[]);
+            move_to(&cfg.failed_dir, zip, &name);
+            return false;
+        }
+    };
+
+    let repo = cfg.repos_root.join(&project);
+
+    {
+        let mut guard = ACTIVE_REPOS.lock().unwrap();
+        let active = guard.get_or_insert_with(HashSet::new);
+        if active.contains(&repo) {
+            ui::skip(&format!(
+                "{name}: skipping, {} is already being processed",
+                project
+            ));
+            return false;
+        }
+        active.insert(repo.clone());
+    }
+
+    ACTIVE_COUNT.fetch_add(1, Ordering::Relaxed);
+
+    let zip = zip.to_path_buf();
+    let cfg_processed = cfg.processed_dir.clone();
+    let cfg_failed = cfg.failed_dir.clone();
+    let name_clone = name.clone();
+    let project_clone = project.clone();
+    let repo_clone = repo.clone();
+
+    std::thread::spawn(move || {
+        crate::ui::start_task(&project_clone);
+
+        let wrapper = match detect_wrapper_prefix(&zip) {
+            Ok(w) => w,
+            Err(e) => {
+                let recovery = recovery_for(&format!("{e:#}"), &project_clone, &zip);
+                ui::error_panel(&name_clone, &format!("{e:#}"), &recovery);
+                move_to(&cfg_failed, &zip, &name_clone);
+                notify(
+                    &format!("ship failed: {}", project_clone),
+                    "Check the terminal for details",
+                    None,
+                    true,
+                );
+                failed.fetch_add(1, Ordering::Relaxed);
+                ACTIVE_REPOS
+                    .lock()
+                    .unwrap()
+                    .as_mut()
+                    .unwrap()
+                    .remove(&repo_clone);
+                ACTIVE_COUNT.fetch_sub(1, Ordering::Relaxed);
+                return;
+            }
+        };
+
+        let started = std::time::Instant::now();
+        match process_zip_inner(&repo_clone, &zip, &name_clone, wrapper.as_deref()) {
+            Ok(outcome) => {
+                let tag = outcome.tag.as_deref().unwrap_or("?");
+                ui::section_ok(
+                    &project_clone,
+                    tag,
+                    outcome.pr_url.as_deref(),
+                    started.elapsed(),
+                );
+                move_to(&cfg_processed, &zip, &name_clone);
+
+                let title = format!("ship succeeded: {}", project_clone);
+                let body = match &outcome.pr_url {
+                    Some(url) => format!("Tag: {}\n{}", tag, url),
+                    None => format!("Tag: {}", tag),
+                };
+                notify(&title, &body, outcome.pr_url.as_deref(), false);
+
+                shipped.fetch_add(1, Ordering::Relaxed);
+            }
+            Err(e) => {
+                let err = format!("{e:#}");
+                let recovery = recovery_for(&err, &project_clone, &zip);
+                ui::error_panel(&name_clone, &err, &recovery);
+                move_to(&cfg_failed, &zip, &name_clone);
+                notify(
+                    &format!("ship failed: {}", project_clone),
+                    "Check the terminal for details",
+                    None,
+                    true,
+                );
+                failed.fetch_add(1, Ordering::Relaxed);
+            }
+        }
+
+        ACTIVE_REPOS
+            .lock()
+            .unwrap()
+            .as_mut()
+            .unwrap()
+            .remove(&repo_clone);
+        ACTIVE_COUNT.fetch_sub(1, Ordering::Relaxed);
+    });
+
+    true
+}
+
+fn recovery_for(err: &str, project: &str, zip: &Path) -> Vec<String> {
+    let zip_name = zip
+        .file_name()
+        .and_then(|n| n.to_str())
+        .unwrap_or("<zip>")
+        .to_string();
+    let repo = format!("~/Documents/Repos/{project}");
+    let failed_dir = "~/Downloads/.failed";
+
+    if err.contains("ship.toml ↔ working tree mismatch")
+        || err.contains("invalid [version].bump")
+        || err.contains("not a conventional subject")
+    {
+        return vec![
+            format!("cd {repo}"),
+            format!("unzip -o {failed_dir}/{zip_name} -d ."),
+            "edit ship.toml (fix the file paths or commit messages)".into(),
+            "ship prepare".into(),
+        ];
+    }
+    if err.contains("is TRACKED") || err.contains("not gitignored") {
+        return vec![
+            format!("cd {repo}"),
+            "git rm --cached .github/pr-body.md  # and/or: ship.toml".into(),
+            format!("mv {failed_dir}/{zip_name} ~/Downloads/"),
+            "ship watch --once".into(),
+        ];
+    }
+    if err.contains("git lock files present") {
+        return vec![
+            format!("cd {repo}"),
+            "ls -la .git/*.lock   # confirm no git process is running".into(),
+            "rm .git/*.lock".into(),
+            format!("mv {failed_dir}/{zip_name} ~/Downloads/"),
+            "ship watch --once --clean-locks".into(),
+        ];
+    }
+    if err.contains("no project matches") {
+        return vec!["check the zip name: it must be <project>.zip or <project>-<slug>.zip".into()];
+    }
+    if err.contains("is not a git repository") || err.contains("not a git repository") {
+        return vec![
+            format!("cd {repo}"),
+            "git init -b main".into(),
+            "git remote add origin <url>".into(),
+            "git fetch origin && git reset --hard origin/main".into(),
+            format!("mv {failed_dir}/{zip_name} ~/Downloads/"),
+            "ship watch --once".into(),
+        ];
+    }
+    if err.contains("not on main") || err.contains("switch to main") {
+        return vec![
+            format!("cd {repo}"),
+            "git switch main".into(),
+            format!("mv {failed_dir}/{zip_name} ~/Downloads/"),
+            "ship watch --once".into(),
+        ];
+    }
+    if err.contains("ahead of") && err.contains("commit") {
+        return vec![
+            format!("cd {repo}"),
+            "git push origin main   # or: git reset --hard origin/main".into(),
+            format!("mv {failed_dir}/{zip_name} ~/Downloads/"),
+            "ship watch --once".into(),
+        ];
+    }
+    if err.contains("tag v") && err.contains("already exists") {
+        return vec![
+            format!("cd {repo}"),
+            "git tag -l   # see what's already there".into(),
+            "manually choose the next version, or delete the stray tag".into(),
+        ];
+    }
+    if err.contains("Operation not permitted") || err.contains("FSEvents") {
+        return vec!["ship watch --poll".into()];
+    }
+    if err.contains("ship.toml") && err.contains("not found") {
+        return vec![
+            "the zip is missing ship.toml — regenerate it with the manifest included".into(),
+        ];
+    }
+    Vec::new()
+}

+fn process_zip_inner(
+    repo: &Path,
+    zip: &Path,
+    name: &str,
+    wrapper: Option<&str>,
+) -> Result<crate::pipeline::Outcome> {
+    let detail = wrapper.map(|w| format!("strip wrapper '{w}/'"));
+    ui::section_begin(name, detail.as_deref());
+
+    let shell = RealShell::new(repo.to_path_buf());
+    if !guards::is_git_repo_root(&shell, repo)? {
+        bail!("{} is not a git repository root", repo.display());
+    }
+
+    (|| -> Result<crate::pipeline::Outcome> {
+        let default = guards::default_branch(&shell, "origin")?;
+        guards::assert_on_main_and_in_sync(&shell, "origin", &default)?;
+        guards::ensure_no_git_locks(&shell, false)?;
+
+        let stashed = guards::is_dirty(&shell)?;
+        if stashed {
+            run_checked(
+                &shell,
+                &[
+                    "git",
+                    "stash",
+                    "push",
+                    "-u",
+                    "-m",
+                    &format!("ship: wip before {}", zip.display()),
+                ],
+            )?;
+            ui::info("WIP stashed");
+        }
+
+        if let Err(e) = run_checked(
+            &shell,
+            &["git", "pull", "--ff-only", "--quiet", "origin", &default],
+        ) {
+            if stashed {
+                let _ = run_checked(&shell, &["git", "stash", "pop"]);
+            }
+            return Err(e.context("git pull --ff-only failed — WIP restored"));
+        }
+
+        // 1. Extract to temp dir to read the manifest
+        let tmp = mktemp_dir()?;
+        run_checked(
+            &shell,
+            &[
+                "unzip",
+                "-q",
+                "-o",
+                &zip.to_string_lossy(),
+                "-d",
+                &tmp.to_string_lossy(),
+            ],
+        )?;
+
+        let source_dir = if let Some(prefix) = wrapper {
+            let p = tmp.join(prefix);
+            if !p.is_dir() {
+                let _ = std::fs::remove_dir_all(&tmp);
+                bail!("expected {}/ inside zip", prefix);
+            }
+            p
+        } else {
+            tmp.clone()
+        };
+
+        let manifest_path = source_dir.join("ship.toml");
+        if !manifest_path.exists() {
+            let _ = std::fs::remove_dir_all(&tmp);
+            bail!("ship.toml is missing from the root of the extracted repo.");
+        }
+        let manifest = Manifest::load(&manifest_path)?;
+
+        // 2. Apply the tree (3-way merge or blind overwrite)
+        if let Some(base) = &manifest.delivery.base_commit {
+            let short_base = if base.len() > 7 { &base[..7] } else { base };
+            ui::info(&format!("3-way merge enabled (base: {})", short_base));
+            let ai_branch = "ship-temp-ai";
+
+            // Create AI snapshot branch
+            run_checked(&shell, &["git", "switch", "-C", ai_branch, base])?;
+            run_checked(&shell, &["git", "rm", "-r", "-f", "."])?;
+            copy_dir_contents(&source_dir, repo)?;
+            run_checked(&shell, &["git", "add", "-A"])?;
+            let _ = shell.run(&["git", "commit", "-m", "AI delivery snapshot"]);
+
+            // Merge into feature branch
+            let branch = &manifest.delivery.branch;
+            run_checked(
+                &shell,
+                &["git", "switch", "-C", branch, &format!("origin/{default}")],
+            )?;
+            let merge = shell.run(&["git", "merge", ai_branch, "--no-commit", "--no-ff"])?;
+            // Unstage the merge so ensure_committed can group the files correctly.
+            let _ = shell.run(&["git", "reset"]);
+            if merge.status != 0 {
+                let _ = shell.run(&["git", "merge", "--abort"]);
+                let _ = shell.run(&["git", "branch", "-D", ai_branch]);
+                let _ = std::fs::remove_dir_all(&tmp);
+                bail!(
+                    "Merge conflict! The AI's zip touches the same lines as your recent pushes.\nPlease resolve manually or regenerate the zip from the latest main."
+                );
+            }
+            let _ = shell.run(&["git", "branch", "-D", ai_branch]);
+
+            // Re-copy untracked config files (ship.toml, pr-body) that were removed by `git rm`
+            let _ = std::fs::copy(source_dir.join("ship.toml"), repo.join("ship.toml"));
+            let _ = std::fs::create_dir_all(repo.join(".github"));
+            let _ = std::fs::copy(
+                source_dir.join(".github/pr-body.md"),
+                repo.join(".github/pr-body.md"),
+            );
+        } else {
+            ui::warn("No base_commit in ship.toml. Blind overwrite. Pushed commits may be lost.");
+            let branch = &manifest.delivery.branch;
+            run_checked(
+                &shell,
+                &["git", "switch", "-C", branch, &format!("origin/{default}")],
+            )?;
+            copy_dir_contents(&source_dir, repo)?;
+        }
+
+        let _ = std::fs::remove_dir_all(&tmp);
+
+        // 3. Revert mode changes
+        let reverted = crate::normalize::exec_bit(&shell)?;
+        if reverted > 0 {
+            ui::warn(&format!(
+                "reverted spurious mode changes on {reverted} tracked file(s)"
+            ));
+        }
+
+        // 4. Prepare & Tag
+        let prep = pipeline::prepare(false, false, false, &manifest, &shell)?;
+        let t = pipeline::tag(false, false, &manifest, &shell)?;
+        let outcome = crate::pipeline::Outcome {
+            pr_url: prep.pr_url.or(t.pr_url),
+            merged: prep.merged.or(t.merged),
+            tag: t.tag,
+        };
+
+        if stashed {
+            if run_checked(&shell, &["git", "stash", "pop"]).is_ok() {
+                ui::info("WIP restored");
+            } else {
+                ui::warn("WIP conflict — stash preserved; resolve in tree, then: git stash drop");
+            }
+        }
+        Ok(outcome)
+    })()
+}
+
+fn mktemp_dir() -> Result<PathBuf> {
+    let out = std::process::Command::new("mktemp").arg("-d").output()?;
+    if !out.status.success() {
+        bail!("mktemp -d failed");
+    }
+    let p = String::from_utf8_lossy(&out.stdout).trim().to_string();
+    if p.is_empty() {
+        bail!("mktemp -d returned empty path");
+    }
+    Ok(PathBuf::from(p))
+}

+fn copy_dir_contents(src: &Path, dest: &Path) -> Result<()> {
+    for entry in std::fs::read_dir(src)? {
+        let entry = entry?;
+        let ft = entry.file_type()?;
+        let sp = entry.path();
+        let dp = dest.join(entry.file_name());
+        if ft.is_dir() {
+            std::fs::create_dir_all(&dp)?;
+            copy_dir_contents(&sp, &dp)?;
+        } else if ft.is_file() {
+            if let Some(parent) = dp.parent() {
+                std::fs::create_dir_all(parent)?;
+            }
+            std::fs::copy(&sp, &dp)
+                .with_context(|| format!("copying {} → {}", sp.display(), dp.display()))?;
+        }
+    }
+    Ok(())
+}

+fn detect_wrapper_prefix(zip: &Path) -> Result<Option<String>> {
+    let entries = zip_entries(zip)?;
+    let normalized: Vec<&str> = entries
+        .iter()
+        .map(|e| e.strip_prefix("./").unwrap_or(e.as_str()))
+        .collect();
+    if normalized
+        .iter()
+        .any(|e| *e == ".git/" || e.contains("/.git/") || e.starts_with(".git/"))
+    {
+        bail!("zip contains .git — refusing to extract");
+    }
+    if normalized.contains(&"ship.toml") {
+        return Ok(None);
+    }
+    let candidates: Vec<&str> = normalized
+        .iter()
+        .filter_map(|e| {
+            e.strip_suffix("/ship.toml")
+                .filter(|p| !p.is_empty() && !p.contains('/'))
+        })
+        .collect();
+    match candidates.len() {
+        0 => {
+            let preview: Vec<String> = normalized
+                .iter()
+                .take(20)
+                .map(|e| format!("    {e}"))
+                .collect();
+            let more = if normalized.len() > 20 {
+                format!("\n    … ({} more)", normalized.len() - 20)
+            } else {
+                String::new()
+            };
+            bail!(
+                "zip has no ship.toml at the archive root or one level deep.\nZip contains {} entries; first 20:\n{}\n{}",
+                normalized.len(),
+                preview.join("\n"),
+                more
+            );
+        }
+        1 => Ok(Some(candidates[0].to_string())),
+        _ => bail!(
+            "zip has multiple ship.toml under different dirs: {:?}",
+            candidates
+        ),
+    }
+}
+
+fn zip_entries(zip: &Path) -> Result<Vec<String>> {
+    let out = std::process::Command::new("unzip")
+        .args(["-Z1", &zip.to_string_lossy()])
+        .output()
+        .context("running unzip -Z1")?;
+    if !out.status.success() {
+        bail!(
+            "unzip -Z1 failed (corrupt or not a zip?): {}",
+            String::from_utf8_lossy(&out.stderr).trim()
+        );
+    }
+    Ok(String::from_utf8_lossy(&out.stdout)
+        .lines()
+        .map(|s| s.trim().to_string())
+        .filter(|s| !s.is_empty() && !s.ends_with('/'))
+        .collect())
+}
+
+fn resolve_project(repos_root: &Path, stem: &str) -> Result<Option<(String, String)>> {
+    let projects = list_projects(repos_root)?;
+    let mut best: Option<(String, String)> = None;
+    for project in &projects {
+        if let Some(slug) = match_project(stem, project) {
+            let replace = match &best {
+                None => true,
+                Some((b, _)) => project.len() > b.len(),
+            };
+            if replace {
+                best = Some((project.clone(), slug));
+            }
+        }
+    }
+    Ok(best)
+}
+
+fn match_project(stem: &str, project: &str) -> Option<String> {
+    if stem == project {
+        return Some(String::new());
+    }
+    let prefix = format!("{project}-");
+    if let Some(slug) = stem.strip_prefix(&prefix) {
+        return Some(slug.to_string());
+    }
+    let stem_lower = stem.to_ascii_lowercase();
+    let project_lower = project.to_ascii_lowercase();
+    if stem_lower == project_lower {
+        return Some(String::new());
+    }
+    let prefix_lower = format!("{project_lower}-");
+    if stem_lower.starts_with(&prefix_lower) {
+        return Some(stem[prefix_lower.len()..].to_string());
+    }
+    None
+}
+
+fn move_to(dir: &Path, zip: &Path, name: &str) {
+    if !zip.exists() {
+        return;
+    }
+    let _ = std::fs::rename(zip, dir.join(name));
+}
+
+fn list_zips(dir: &Path) -> Result<Vec<PathBuf>> {
+    let mut out = Vec::new();
+    for e in std::fs::read_dir(dir)?.flatten() {
+        if e.file_type().map(|t| t.is_file()).unwrap_or(false) && is_zip(&e.path()) {
+            out.push(e.path());
+        }
+    }
+    Ok(out)
+}
+
+fn is_zip(p: &Path) -> bool {
+    p.extension()
+        .and_then(|s| s.to_str())
+        .map(|s| s.eq_ignore_ascii_case("zip"))
+        .unwrap_or(false)
+}
+
+fn notify(title: &str, body: &str, pr_url: Option<&str>, is_error: bool) {
+    let sound = if is_error { "Basso" } else { "Glass" };
+
+    #[cfg(target_os = "macos")]
+    {
+        // Check if terminal-notifier is installed for clickable notifications
+        let has_tn = std::process::Command::new("which")
+            .arg("terminal-notifier")
+            .output()
+            .map(|o| o.status.success())
+            .unwrap_or(false);
+
+        if has_tn {
+            let mut cmd = std::process::Command::new("terminal-notifier");
+            cmd.args([
+                "-title",
+                title,
+                "-message",
+                body,
+                "-sound",
+                sound,
+                "-subtitle",
+                "ship pipeline",
+            ]);
+            if let Some(url) = pr_url {
+                cmd.args(["-open", url]);
+            }
+            let _ = cmd.output();
+        } else {
+            // Fallback to AppleScript
+            let script = format!(
+                r#"display notification {:?} with title {:?} subtitle "ship pipeline" sound name {:?}"#,
+                body, title, sound
+            );
+            let _ = std::process::Command::new("osascript")
+                .args(["-e", &script])
+                .output();
+        }
+    }
+    #[cfg(target_os = "linux")]
+    {
+        let mut cmd = std::process::Command::new("notify-send");
+        if is_error {
+            cmd.args(["-u", "critical"]);
+        }
+        cmd.args([title, body]);
+        let _ = cmd.output();
+    }
+    #[cfg(not(any(target_os = "macos", target_os = "linux")))]
+    {
+        let _ = (title, body, pr_url, is_error);
+    }
+}
+
+struct PidLock {
+    path: PathBuf,
+}
+impl Drop for PidLock {
+    fn drop(&mut self) {
+        let _ = std::fs::remove_file(&self.path);
+    }
+}
+
+fn acquire_lock(path: &Path) -> Result<PidLock> {
+    if let Ok(s) = std::fs::read_to_string(path)
+        && let Ok(pid) = s.trim().parse::<u32>()
+        && process_alive(pid)
+    {
+        bail!("another ship watch is running (pid {pid})");
+    }
+    std::fs::write(path, std::process::id().to_string())?;
+    Ok(PidLock {
+        path: path.to_path_buf(),
+    })
+}
+
+fn process_alive(pid: u32) -> bool {
+    std::process::Command::new("kill")
+        .args(["-0", &pid.to_string()])
+        .status()
+        .map(|s| s.success())
+        .unwrap_or(false)
+}
